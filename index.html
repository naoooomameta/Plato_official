} } // ========================================================= // ① 問い合わせ受付 // ========================================================= function handleContact(e) { const p = e.parameter; const name = p.name || ''; } catch (mailErr) { Logger.log('管理者メール通知エラー: ' + mailErr.message); } // ── ユーザーへ①サンクスメール ── if (email) { try { MailApp.sendEmail({ to: email, subject: '【プラトン】お問い合わせありがとうございます', body: (name ? name + ' 様\n\n' : '') + 'この度は、文系の大学受験専門塾プラトンへ\n' + 'お問い合わせいただき、誠にありがとうございます。\n\n' + 'お問い合わせ内容を確認のうえ、\n' + '2〜3営業日以内に担当の佐野よりご連絡いたします。\n\n' + 'なお、このあと表示される画面から\n' + '事前に面談のご希望日時をお選びいただくと、\n' + 'よりスムーズにご案内が可能です。\n\n' + '━━━━━━━━━━━━━━━━━━━━\n' + '■ お問い合わせ内容\n' + '━━━━━━━━━━━━━━━━━━━━\n' + 'お名前: ' + name + '\n' + 'メール: ' + email + '\n' + '電話番号: ' + tel + '\n' + '学年: ' + grade + '\n' + '志望大学: ' + university + '\n' + 'ご相談内容: ' + message + '\n' + '━━━━━━━━━━━━━━━━━━━━\n\n' + 'ご不明な点がございましたら、\n' + '下記までお気軽にご連絡ください。\n' + SIGNATURE }); } catch (userMailErr) { Logger.log('ユーザーメール送信エラー: ' + userMailErr.message); } } return jsonResponse({ ok: true }); } // ========================================================= // ② 面談予約登録 // ========================================================= function handleBooking(e) { const p = e.parameter; const slot = JSON.parse(p.slot || '{}'); const name = p.name || ''; const email = p.email || ''; const meeting = p.meeting || ''; const topics = e.parameters.topics || []; const note = p.note || ''; // ── Googleカレンダーにイベント作成 ── const startTime = new Date(slot.startIso || slot.start); const endTime = new Date(startTime.getTime() + SLOT_DURATION_MIN * 60 * 1000); const cal = CalendarApp.getCalendarById(CALENDAR_ID); if (!cal) throw new Error('カレンダーが見つかりません'); const title = '【面談予約】' + (name || '名前未入力'); const description = [ 'お名前: ' + name, 'メール: ' + email, '連絡手段: ' + meeting, '聞きたいこと: ' + topics.join(', '), '伝言: ' + note ].join('\n'); cal.createEvent(title, startTime, endTime, { description: description, guests: email || undefined, sendInvites: !!email }); // ── スプレッドシートに記録 ── if (SHEET_ID) { try { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_BOOKING); if (!sheet) { sheet = ss.insertSheet(SHEET_BOOKING); sheet.appendRow(['日時', 'お名前', 'メール', '連絡手段', '聞きたいこと', '伝言', '登録日時']); } sheet.appendRow([ Utilities.formatDate(startTime, TIMEZONE, 'yyyy/MM/dd HH:mm'), name, email, meeting, topics.join(', '), note, Utilities.formatDate(new Date(), TIMEZONE, 'yyyy/MM/dd HH:mm:ss') ]); } catch (sheetErr) { Logger.log('予約シート記録エラー: ' + sheetErr.message); } } // ── 管理者へメール通知 ── try { MailApp.sendEmail({ to: CALENDAR_ID, subject: '【プラトン】新規面談予約: ' + name, body: '新しい面談予約が入りました。\n\n' + '日時: ' + slot.dateLabel + ' ' + slot.timeLabel + '\n' + description + '\n\n' + 'Googleカレンダーに自動登録済みです。' }); } catch (mailErr) { Logger.log('管理者メール通知エラー: ' + mailErr.message); } // ── ユーザーへ②予約確定メール ── if (email) { try { MailApp.sendEmail({ to: email, subject: '【プラトン】面談日程が確定しました', body: (name ? name + ' 様\n\n' : '') + '面談のご予約ありがとうございます。\n' + '下記の日程で、担当の佐野よりご連絡させていただきます。\n\n' + '━━━━━━━━━━━━━━━━━━━━\n' + '■ 面談日時\n' + ' ' + slot.dateLabel + ' ' + slot.timeLabel + '\n\n' + '■ 連絡方法\n' + ' ' + (meeting || '未選択') + '\n\n' + '■ 所要時間\n' + ' 約30分\n' + '━━━━━━━━━━━━━━━━━━━━\n\n' + '当日は、以下のようなご相談に対応いたします。\n\n' + '・お子様に合った学習スタイルについて\n' + '・プラトンの指導内容と他塾との違い\n' + '・具体的な合格実績とその学習プロセス\n' + '・料金体系と入塾までの流れ\n\n' + '無理な勧誘は一切いたしませんので、\n' + 'お気軽にお話しください。\n\n' + '━━━━━━━━━━━━━━━━━━━━\n' + '■ 日程変更・キャンセルについて\n' + '━━━━━━━━━━━━━━━━━━━━\n' + 'ご都合が悪くなった場合は、\n' + '下記メールアドレスまでご連絡ください。\n' + '別日への変更も承ります。\n' + SIGNATURE }); } catch (userMailErr) { Logger.log('ユーザーメール送信エラー: ' + userMailErr.message); } } return jsonResponse({ ok: true }); } // ========================================================= // 空き枠取得ロジック // ========================================================= function getAvailableSlots(daysAhead, limit) { const cal = CalendarApp.getCalendarById(CALENDAR_ID); if (!cal) throw new Error('カレンダーが見つかりません: ' + CALENDAR_ID); const now = new Date(); const start = new Date(now); start.setHours(0, 0, 0, 0); const end = new Date(start); end.setDate(end.getDate() + daysAhead); const events = cal.getEvents(start, end); const busy = events.map(ev => ({ start: ev.getStartTime().getTime(), end: ev.getEndTime().getTime() })); const slots = []; const weekdays = ['日', '月', '火', '水', '木', '金', '土']; for (let d = 0; d < daysAhead; d++) { const day = new Date(start); day.setDate(day.getDate() + d); for (let h = START_HOUR; h < END_HOUR; h++) { const stepsPerHour = Math.floor(60 / SLOT_DURATION_MIN); for (let step = 0; step < stepsPerHour; step++) { const slotStart = new Date(day); slotStart.setHours(h, step * SLOT_DURATION_MIN, 0, 0); const slotEnd = new Date(slotStart.getTime() + (SLOT_DURATION_MIN + BUFFER_MIN) * 60 * 1000); const dayEnd = new Date(day); dayEnd.setHours(END_HOUR, 0, 0, 0); if (slotEnd.getTime() > dayEnd.getTime()) continue; if (slotStart.getTime() < now.getTime() + 60 * 60 * 1000) continue; const isBusy = busy.some(b => slotStart.getTime() < b.end && slotEnd.getTime() > b.start ); if (isBusy) continue; const mm = ('0' + (slotStart.getMonth() + 1)).slice(-2); const dd = ('0' + slotStart.getDate()).slice(-2); const ww = weekdays[slotStart.getDay()]; const hh = ('0' + slotStart.getHours()).slice(-2); const mi = ('0' + slotStart.getMinutes()).slice(-2); slots.push({ dateLabel: mm + '/' + dd + '（' + ww + '）', timeLabel: hh + ':' + mi + '〜', startIso: slotStart.toISOString(), start: slotStart.toISOString() }); if (limit > 0 && slots.length >= limit) { return slots; } } } } return slots; } // ===== ユーティリティ ===== function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON);
